4.2 Node-RED 配置与流

Node-RED 是控制中枢。它通过 node-red-contrib-s7 节点与西门子 PLC 通信，通过 MQTT 节点与 Telegraf 对接。

需要的 Node-RED 包：

# 进入 Node-RED 容器安装
docker exec -it <nodered_container> bash
npm install node-red-contrib-s7
npm install node-red-contrib-influxdb
npm install node-red-dashboard
npm install @vegarwe/node-red-contrib-http-request  # HTTP API 用于 Grafana 写回

Node-RED 流文件位置： ./nodered/data/flows.json

核心包含 3 个流：

流名称作用核心节点
flow-plc-read 定时读取 PLC 数据 → MQTT 发布 S7 in (定期) → mqtt out
flow-plc-write 接收 HTTP 控制请求 → 写入 PLC http in → S7 out → http response
flow-heartbeat 网关心跳监控 inject (30s) → mqtt out

5. 核心：实现"从 Grafana 写回 PLC"

这是 Trioop 最值钱的能力。Grafana 原生只支持读，不支持写。 以下是三种实现方案：

5.1 方案一：Grafana 自定义 Panel 插件（推荐）

创建一个 Grafana Panel Plugin，在 React 组件里放按钮，点击后 fetch 调用 Node-RED API。

插件目录： grafana/plugins/plc-control-panel/

// src/PLControlPanel.tsx — 核心组件
import React, { useState } from 'react';
import { PanelProps } from '@grafana/data';

interface Props extends PanelProps {}
interface ControlTarget {
  name: string;
  address: string;   // 如 "DB1,REAL4"
  type: 'float' | 'bool' | 'int';
  currentValue: number;
}

export const PLControlPanel: React.FC<Props> = ({ data, options }) => {
  const controls: ControlTarget[] = options.controls || [];
  const [values, setValues] = useState<Record<string, number>>({});
  const [loading, setLoading] = useState<Record<string, boolean>>({});

  const handleWrite = async (target: ControlTarget, value: number) => {
    setLoading(prev => ({ ...prev, [target.name]: true }));
    try {
      const resp = await fetch('http://gateway-ip:1880/api/plc/write', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          address: target.address,
          type: target.type,
          value: value
        })
      });
      if (!resp.ok) throw new Error(`Write failed: ${resp.status}`);
    } catch (err) {
      console.error('PLC write error:', err);
    } finally {
      setLoading(prev => ({ ...prev, [target.name]: false }));
    }
  };

  return (
    <div style={{ padding: 8 }}>
      {controls.map(target => (
        <div key={target.name} style={{
          display: 'flex', alignItems: 'center',
          justifyContent: 'space-between',
          padding: '8px 12px', margin: '4px 0',
          background: '#1a1a2e', borderRadius: 6
        }}>
          <span>{target.name}</span>
          <div style={{ display: 'flex', gap: 8, alignItems: 'center' }}>
            {target.type === 'bool' ? (
              <button onClick={() => handleWrite(target, values[target.name] ? 0 : 1)}
                style={{
                  padding: '6px 16px',
                  background: values[target.name] ? '#4caf50' : '#666',
                  color: '#fff', border: 'none', borderRadius: 4,
                  cursor: 'pointer'
                }}>
                {values[target.name] ? 'ON' : 'OFF'}
              </button>
            ) : (
              <>
                <input type="number" step="0.1"
                  value={values[target.name] ?? target.currentValue}
                  onChange={e => setValues(prev => ({
                    ...prev, [target.name]: parseFloat(e.target.value)
                  }))}
                  style={{ width: 80, padding: 4, background: '#0a0a14',
                    color: '#fff', border: '1px solid #333', borderRadius: 4 }}
                />
                <button onClick={() => handleWrite(target, values[target.name])}
                  disabled={loading[target.name]}
                  style={{
                    padding: '6px 16px',
                    background: loading[target.name] ? '#666' : '#2196f3',
                    color: '#fff', border: 'none', borderRadius: 4,
                    cursor: 'pointer'
                  }}>
                  {loading[target.name] ? '⏳' : '写入'}
                </button>
              </>
            )}
          </div>
        </div>
      ))}
    </div>
  );
};

5.3 方案三：独立 C# Web 控制服务

如果你希望和现有的 TEST_101 C# 项目结合，可以写一个 ASP.NET Core Web API 作为控制服务：

// PlcControlService/Controllers/PlcController.cs
[ApiController]
[Route("api/plc")]
public class PlcController : ControllerBase
{
    private readonly Snap7Client _client = new();

    [HttpPost("write")]
    public IActionResult Write([FromBody] WriteRequest request)
    {
        var result = _client.ConnectTo("192.168.1.100", 0, 1);
        if (result != 0)
            return StatusCode(502, "PLC 连接失败");

        byte[] buffer = request.Type switch
        {
            "float" => BitConverter.GetBytes((float)request.Value),
            "int"   => BitConverter.GetBytes((int)request.Value),
            "bool"  => new[] { (byte)(request.Value > 0 ? 1 : 0) },
            _ => throw new ArgumentException("Unknown type")
        };

        // 解析地址 "DB1,REAL4" → DB=1, offset=4
        var (dbNum, offset) = ParseAddress(request.Address);
        _client.DBWrite(dbNum, offset, buffer);
        _client.Disconnect();

        return Ok(new { success = true });
    }
}

public record WriteRequest(
    string Address,  // "DB1,REAL4"
    string Type,     // "float" | "int" | "bool"
    double Value
);

7.1 Node-RED S7 读写流（函数节点核心代码）

在 Node-RED 中，Function 节点里写 JavaScript 处理数据转换：

// Node-RED Function: "格式化为 InfluxDB 可读 JSON"
// 输入: msg.payload = { "设备运行状态": true, "温度_1": 25.3, "压力_1": 0.6 }

const result = {
  device_id: "S7-1200_01",
  plc_type: "S7-1200",
  timestamp: new Date().toISOString(),
  temperature: msg.payload["温度_1"],
  pressure: msg.payload["压力_1"],
  speed: msg.payload["电机转速"],
  is_running: msg.payload["设备运行状态"]
};

// 附加报警逻辑：温度超标自动标记
if (result.temperature > 80.0) {
  result.alarm = "高温预警";
  result.severity = 2;
}

msg.payload = result;
return msg;

// Node-RED Function: "写回前校验与转换"
// 输入: msg.payload = { address: "DB1,REAL4", type: "float", value: 35.5 }

// 地址格式校验
const match = msg.payload.address.match(/^DB(\d+),(REAL|INT|BOOL|X)(\d+(?:\.\d+)?)$/);
if (!match) {
  msg.payload = { error: "地址格式错误: " + msg.payload.address };
  return [null, msg];  // 发到错误输出
}

// 值范围校验
if (msg.payload.type === "float") {
  const val = Number(msg.payload.value);
  if (val < -50 || val > 200) {
    msg.payload = { error: "温度值超出合理范围: " + val };
    return [null, msg];
  }
}

return [msg, null];  // 校验通过，写入主流程

7.2 C# Snap7 控制服务关键函数

// Snap7Helper.cs — C# 连接西门子 PLC 工具类
using Snap7;

public class Snap7Helper : IDisposable
{
    private readonly S7Client _client = new();

    /// 连接 PLC
    /// 返回 (成功否, 错误消息)
    public (bool Success, string? Error) Connect(
        string ip, int rack = 0, int slot = 1)
    {
        int result = _client.ConnectTo(ip, rack, slot);
        if (result == 0)
            return (true, null);
        return (false, _client.ErrorText(result));
    }

    /// 读取 DB 块中的浮点数
    public (bool, float) ReadFloat(int dbNumber, int byteOffset)
    {
        byte[] buffer = new byte[4];
        int result = _client.DBRead(dbNumber, byteOffset, 4, buffer);
        if (result == 0)
            return (true, BitConverter.ToSingle(buffer, 0));
        return (false, 0);
    }

    /// 写入浮点数到 DB 块
    public (bool, string?) WriteFloat(int dbNumber, int byteOffset, float value)
    {
        byte[] buffer = BitConverter.GetBytes(value);
        int result = _client.DBWrite(dbNumber, byteOffset, buffer);
        if (result == 0)
            return (true, null);
        return (false, _client.ErrorText(result));
    }

    /// 写入布尔值（位）到 DB 块
    public (bool, string?) WriteBool(int dbNumber, int byteOffset, int bitIndex, bool value)
    {
        byte[] buffer = new byte[1];
        // 先读出当前字节
        _client.DBRead(dbNumber, byteOffset, 1, buffer);
        // 修改指定位
        if (value)
            buffer[0] |= (byte)(1 << bitIndex);
        else
            buffer[0] &= (byte)~(1 << bitIndex);
        // 写回
        int result = _client.DBWrite(dbNumber, byteOffset, buffer);
        if (result == 0)
            return (true, null);
        return (false, _client.ErrorText(result));
    }

    public void Disconnect() => _client.Disconnect();
    public void Dispose() => _client.Dispose();
}

"我实现了一套基于开源技术的工业物联网监控系统，采用 MING 栈：MQTT（Mosquitto）+ InfluxDB + Node-RED + Grafana。

Node-RED 通过 S7 协议（ISO-on-TCP port 102）读取西门子 PLC 的 DB 块数据，经 MQTT 发布后由 Telegraf 写入时序数据库。Grafana 上我开发了一个自定义 Panel 插件，用 React 组件实现了从仪表盘直接写回 PLC 的控制按钮——这是 Trioop 这类商业方案的核心能力。

部署方面全部 Docker Compose 编排，硬件选用香橙派（199 元）作为边缘网关，数据本地处理，支持断线缓冲。整个系统的监控延迟在 ms 级，控制指令从点击到 PLC 响应 < 50ms。

通讯层我熟悉 Snap7 和 s7netplus 两个 C# 库的读写操作，包括 DB 块的定点/浮点读写、位操作、错误码解析。也了解西门子 S7-1200/1500 需要开启 PUT/GET 访问和取消优化块访问这两个关键配置。"
